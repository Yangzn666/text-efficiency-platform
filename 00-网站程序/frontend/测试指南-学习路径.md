**预期结果：**

#### ✅ 手动创建复习计划（通过控制台）
打开浏览器控制台，执行以下代码：

```javascript
// 导入store
import { useLearningPathStore } from '@/stores/learningPath'
const store = useLearningPathStore()

// 创建复习计划
store.createReviewSchedule(
  'test_001',
  '极限与连续',
  '数学一',
  new Date().toISOString().split('T')[0]
)

// 查看今日复习
console.log('今日复习:', store.getTodayReviews())

// 查看待复习
console.log('待复习:', store.getPendingReviews())
```

**预期结果：**
- 复习计划已创建
- 今日复习列表显示新创建的知识点
- 记忆保持率显示为100%

#### ✅ 完成复习
1. 在"今日复习任务"区域
2. 点击"完成复习"按钮

**预期结果：**

### 4. 掌握度追踪测试

#### ✅ 更新掌握度记录（通过控制台）

## 📊 测试数据准备脚本

在浏览器控制台中运行以下脚本快速准备测试数据：

```javascript
import { useLearningPathStore } from '@/stores/learningPath'
const store = useLearningPathStore()

const today = new Date().toISOString().split('T')[0]

// 创建多个复习计划
const testPlans = [
  { id: 'math_001', name: '极限与连续', subject: '数学一', daysAgo: 0 },
  { id: 'math_002', name: '导数与微分', subject: '数学一', daysAgo: 1 },
  { id: 'math_003', name: '积分计算', subject: '数学一', daysAgo: 2 },
  { id: 'english_001', name: '核心词汇Unit1', subject: '英语一', daysAgo: 0 },
  { id: 'english_002', name: '长难句分析', subject: '英语一', daysAgo: 3 },
]

testPlans.forEach(plan => {
  const learnDate = new Date()
  learnDate.setDate(learnDate.getDate() - plan.daysAgo)
  
  store.createReviewSchedule(
    plan.id,
    plan.name,
    plan.subject,
    learnDate.toISOString().split('T')[0]
  )
})

// 创建掌握度记录
const masteryData = [
  { id: 'math_001', name: '极限', subject: '数学一', scores: [60, 65, 70] },
  { id: 'math_002', name: '导数', subject: '数学一', scores: [80, 85, 90] },
  { id: 'math_003', name: '积分', subject: '数学一', scores: [50, 45, 40] },
  { id: 'english_001', name: '词汇', subject: '英语一', scores: [70, 75, 80] },
  { id: 'english_002', name: '语法', subject: '英语一', scores: [55, 60, 58] },
]

masteryData.forEach(data => {
  data.scores.forEach((score, index) => {
    setTimeout(() => {
      store.updateMasteryRecord(data.id, data.name, data.subject, score)
    }, index * 100)
  })
})

console.log('✅ 测试数据创建完成！')
console.log('今日复习:', store.getTodayReviews().length)
console.log('待复习:', store.getPendingReviews().length)
console.log('薄弱知识点:', store.getWeakKnowledgePoints().length)
```
